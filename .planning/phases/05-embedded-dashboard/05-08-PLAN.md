Front-end contract this plan creates (later screens rely on it):

```js
// js/theme.js — classic script in <head>; localStorage key 'ab.theme' = 'system' | 'dark' | 'light' (default 'system')
window.ABTheme = { mode() {}, set(mode) {}, apply() {} };   // apply() toggles class 'dark' on <html>

// js/app.js — deferred, before Alpine
window.AB = {
  route: { name: 'overview', params: {} },
  profile() {},            // 'all' | 'work' | 'personal' from localStorage 'ab.profile', default 'all'
  setProfile(p) {},        // stores, then dispatches window event 'ab:profile'
  profileQuery() {},       // '' or 'profile=work'
  api(method, path, body) {},   // Promise<{ ok, status, data }>; JSON body ('{}' when absent) + Content-Type for POST/PUT; 401 → sessionLost
  every(ms, fn) {},        // setInterval cleared automatically when the route changes
  onLeave(fn) {},          // callback run once before the next screen mounts (close EventSource etc.)
  onReady(name, factory) {},    // Alpine.data(name, factory) now if Alpine is running, else on 'alpine:init'
  fmtNum(n) {}, fmtDuration(ms) {}, fmtTime(ms) {}, fmtCost(usd) {}, stepClass(state) {}, profileClass(p) {}
};
// screen fragment root: <section data-screen="NAME" x-data="NAME" :data-loaded="loadedAttr"> where loadedAttr is 'yes' after the first successful load
```
