```rust
// stats.rs (added)
#[derive(Debug, Clone, serde::Serialize, serde::Deserialize)]
pub struct DayStats { pub day: String, #[serde(flatten)] pub totals: Totals }          // day = "YYYY-MM-DD"
#[derive(Debug, Clone, serde::Serialize, serde::Deserialize)]
pub struct ProfileStats { pub profile: String, #[serde(flatten)] pub totals: Totals }
pub fn day_key<Tz: chrono::TimeZone>(ms: i64, tz: &Tz) -> String;
pub fn query_by_day_in<Tz: chrono::TimeZone>(conn: &Connection, profile: Option<&str>, since_ms: Option<i64>, tz: &Tz) -> BrainResult<Vec<DayStats>>;
pub fn query_by_day(conn: &Connection, profile: Option<&str>, since_ms: Option<i64>) -> BrainResult<Vec<DayStats>>;   // = query_by_day_in(.., &chrono::Local)
pub fn query_by_profile(conn: &Connection, profile: Option<&str>, since_ms: Option<i64>) -> BrainResult<Vec<ProfileStats>>;

// dashboard/api_stats.rs
pub fn routes() -> axum::Router<Arc<AppState>>;   // GET /api/stats
```
