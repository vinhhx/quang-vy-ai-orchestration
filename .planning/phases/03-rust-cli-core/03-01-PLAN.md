Rust contract this plan creates (later plans import it exactly):

```rust
// agent-brain/src/error.rs
pub const EXIT_OK: i32 = 0;
pub const EXIT_ERROR: i32 = 1;
pub const EXIT_ROUTING: i32 = 2;
pub const EXIT_PROTOCOL: i32 = 3;
pub const EXIT_BLOCKED: i32 = 4;
pub const EXIT_INTERRUPTED: i32 = 130;
#[derive(Debug, Clone, PartialEq, Eq)]
pub struct BrainError { pub message: String, pub code: i32 }
impl BrainError {
    pub fn new(message: impl Into<String>, code: i32) -> Self;
    pub fn error(message: impl Into<String>) -> Self;      // code EXIT_ERROR
    pub fn not_available(command: &str) -> Self;           // "Lệnh '<command>' chưa có trong bản này", EXIT_ERROR
}
impl std::fmt::Display for BrainError {}                    // writes `message` only
impl std::error::Error for BrainError {}
impl From<std::io::Error> for BrainError {}                 // "Lỗi I/O: <e>", EXIT_ERROR
pub type BrainResult<T> = Result<T, BrainError>;
```
